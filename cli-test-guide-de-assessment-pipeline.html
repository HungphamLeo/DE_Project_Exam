🔎 Phase 3 — Verify dữ liệu trong Postgres

3.1 Dùng psql trong Docker

# Kết nối vào postgres container
docker exec -it <postgres_container_name> psql -U de -d assessment

3.2 SQL kiểm tra nhanh

-- Row counts tất cả bảng staging
SELECT schemaname, tablename,
       n_live_tup AS row_count
FROM pg_stat_user_tables
WHERE schemaname = 'staging'
ORDER BY tablename;
-- Kiểm tra fact_trips grain (không duplicate event_id)
SELECT COUNT(*) total, COUNT(DISTINCT event_id) unique_ids
FROM staging.fact_trips;
-- Kiểm tra SCD2 dim_zone (chỉ 1 current per zone_id)
SELECT zone_id, COUNT(*) cnt
FROM staging.dim_zone
WHERE is_current = TRUE
GROUP BY zone_id
HAVING COUNT(*) > 1;  -- phải ra 0 rows
-- Sample fact_trips với dim joins
SELECT
    f.event_id, f.event_timestamp,
    v.vendor_name, e.event_type_name, p.payment_method_name,
    f.total_value
FROM staging.fact_trips f
LEFT JOIN staging.dim_vendor v       ON f.vendor_key = v.vendor_key
LEFT JOIN staging.dim_event_type e   ON f.event_type_key = e.event_type_key
LEFT JOIN staging.dim_payment_method p ON f.payment_method_key = p.payment_method_key
LIMIT 5;
